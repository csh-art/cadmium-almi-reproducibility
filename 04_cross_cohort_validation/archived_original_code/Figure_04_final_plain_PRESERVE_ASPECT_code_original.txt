# ============================================================
# Figure 04 final composite — preserve aspect ratios
# A/B, C/D, full-width E
# E is NOT stretched or vertically compressed.
# ============================================================

make_figure4_composite_preserve_aspect <- function () 
{
    options(stringsAsFactors = FALSE, warn = 1, scipen = 999)
    graphics.off()
    if (!requireNamespace("png", quietly = TRUE)) {
        cat("检测到缺少R包 png，正在自动安装……\n")
        install.packages("png", repos = "https://cloud.r-project.org", dependencies = TRUE)
    }
    if (!requireNamespace("png", quietly = TRUE)) {
        stop("png 软件包自动安装失败。请把完整安装报错发给我。", call. = FALSE)
    }
    has_ragg <- requireNamespace("ragg", quietly = TRUE)
    project_dir <- paste0("E:/1Data/Cd_Sarcopenia_NHANES/new_workspace/", "03_results/", "M04_HubGenes_cytoHubba_GSE111016_GSE111010")
    figure_dir <- file.path(project_dir, "03_outputs_figures_tables", "figure")
    input_files <- c(A = file.path(figure_dir, "Figure_04A_cytoHubba_consensus_matrix_plain.png"), B = file.path(figure_dir, "Figure_04B_GSE111016_hub_gene_effect_size_forest_plain.png"), C = file.path(figure_dir, "Figure_04C_GSE111016_robust_hub_gene_expression_heatmap_plain.png"), D = file.path(figure_dir, "Figure_04D_pathway_concordance_plain.png"), E = file.path(figure_dir, "Figure_04E_multicohort_forest_plain.png"))
    missing_files <- input_files[!file.exists(input_files)]
    if (length(missing_files) > 0) {
        stop(paste0("缺少以下输入图片：\n", paste0("  - ", names(missing_files), ": ", missing_files, collapse = "\n")), call. = FALSE)
    }
    output_png <- file.path(figure_dir, "Figure_04_final_plain_PRESERVE_ASPECT.png")
    output_pdf <- file.path(figure_dir, "Figure_04_final_plain_PRESERVE_ASPECT.pdf")
    output_code <- file.path(figure_dir, "Figure_04_final_plain_PRESERVE_ASPECT_code.txt")
    panel_images <- lapply(input_files, png::readPNG)
    image_ratio <- vapply(panel_images, function(img) {
        dims <- dim(img)
        dims[2]/dims[1]
    }, numeric(1))
    plot_width_in <- 16
    top_row_1_height_in <- 5.35
    top_row_2_height_in <- 5.45
    bottom_E_height_in <- 8.45
    outer_top_in <- 0.18
    outer_bottom_in <- 0.18
    outer_left_in <- 0.16
    outer_right_in <- 0.16
    horizontal_gap_in <- 0.15
    vertical_gap_1_in <- 0.12
    vertical_gap_2_in <- 0.12
    plot_height_in <- (outer_top_in + top_row_1_height_in + vertical_gap_1_in + top_row_2_height_in + vertical_gap_2_in + bottom_E_height_in + outer_bottom_in)
    column_widths <- c(1, 1)
    panel_label_fontsize <- 18
    draw_panel_preserve_aspect <- function(img, label) {
        cell_width_in <- grid::convertWidth(grid::unit(1, "npc"), "in", valueOnly = TRUE)
        cell_height_in <- grid::convertHeight(grid::unit(1, "npc"), "in", valueOnly = TRUE)
        dims <- dim(img)
        img_ratio <- dims[2]/dims[1]
        cell_ratio <- cell_width_in/cell_height_in
        if (img_ratio >= cell_ratio) {
            raster_width_npc <- 1
            raster_height_npc <- cell_ratio/img_ratio
        }
        else {
            raster_height_npc <- 1
            raster_width_npc <- img_ratio/cell_ratio
        }
        grid::grid.rect(gp = grid::gpar(fill = "white", col = NA))
        grid::grid.raster(image = img, x = 0.5, y = 0.5, width = grid::unit(raster_width_npc, "npc"), height = grid::unit(raster_height_npc, "npc"), interpolate = FALSE)
        grid::grid.text(label = label, x = grid::unit(0.012, "npc"), y = grid::unit(0.992, "npc"), just = c("left", "top"), gp = grid::gpar(fontsize = panel_label_fontsize, fontface = "bold", col = "black"))
    }
    draw_complete_figure <- function() {
        grid::grid.newpage()
        main_layout <- grid::grid.layout(nrow = 5, ncol = 3, heights = grid::unit(c(top_row_1_height_in, vertical_gap_1_in, top_row_2_height_in, vertical_gap_2_in, bottom_E_height_in), "in"), widths = grid::unit(c(1, horizontal_gap_in, 1), c("null", "in", "null")))
        grid::pushViewport(grid::viewport(x = grid::unit(outer_left_in, "in"), y = grid::unit(outer_bottom_in, "in"), width = grid::unit(plot_width_in - outer_left_in - outer_right_in, "in"), height = grid::unit(plot_height_in - outer_top_in - outer_bottom_in, "in"), just = c("left", "bottom"), layout = main_layout))
        grid::pushViewport(grid::viewport(layout.pos.row = 1, layout.pos.col = 1))
        draw_panel_preserve_aspect(panel_images$A, "A")
        grid::popViewport()
        grid::pushViewport(grid::viewport(layout.pos.row = 1, layout.pos.col = 3))
        draw_panel_preserve_aspect(panel_images$B, "B")
        grid::popViewport()
        grid::pushViewport(grid::viewport(layout.pos.row = 3, layout.pos.col = 1))
        draw_panel_preserve_aspect(panel_images$C, "C")
        grid::popViewport()
        grid::pushViewport(grid::viewport(layout.pos.row = 3, layout.pos.col = 3))
        draw_panel_preserve_aspect(panel_images$D, "D")
        grid::popViewport()
        grid::pushViewport(grid::viewport(layout.pos.row = 5, layout.pos.col = 1:3))
        draw_panel_preserve_aspect(panel_images$E, "E")
        grid::popViewport()
        grid::popViewport()
    }
    if (has_ragg) {
        ragg::agg_png(filename = output_png, width = plot_width_in, height = plot_height_in, units = "in", res = 600, background = "white")
    }
    else {
        png_args <- list(filename = output_png, width = plot_width_in, height = plot_height_in, units = "in", res = 600, bg = "white")
        if (capabilities("cairo")) {
            png_args$type <- "cairo"
        }
        do.call(grDevices::png, png_args)
    }
    draw_complete_figure()
    grDevices::dev.off()
    grDevices::cairo_pdf(filename = output_pdf, width = plot_width_in, height = plot_height_in, bg = "white")
    draw_complete_figure()
    grDevices::dev.off()
    function_code <- deparse(make_figure4_composite_preserve_aspect, width.cutoff = 500L)
    function_code[1] <- paste0("make_figure4_composite_preserve_aspect <- ", function_code[1])
    complete_code <- c("# ============================================================", "# Figure 04 final composite — preserve aspect ratios", "# A/B, C/D, full-width E", "# E is NOT stretched or vertically compressed.", "# ============================================================", "", function_code, "", "figure4_result <- make_figure4_composite_preserve_aspect()", "")
    writeLines(complete_code, con = output_code, useBytes = TRUE)
    output_files <- c(PNG = output_png, PDF = output_pdf, Code_TXT = output_code)
    output_status <- data.frame(Output = names(output_files), File = unname(output_files), Exists = file.exists(output_files), Size_MB = round(file.info(output_files)$size/1024^2, 3), stringsAsFactors = FALSE)
    if (!all(output_status$Exists)) {
        stop("至少一个 Figure 4 输出文件没有成功生成。", call. = FALSE)
    }
    cat("\n============================================================\n")
    cat("FIGURE 4 COMPOSITE COMPLETED — ASPECT RATIO PRESERVED\n")
    cat("============================================================\n")
    cat("\nInput image aspect ratios (width / height):\n")
    print(data.frame(Panel = names(image_ratio), Aspect_ratio = round(image_ratio, 3), stringsAsFactors = FALSE), row.names = FALSE)
    cat("\nFinal layout:\n")
    cat("Row 1: A | B\n")
    cat("Row 2: C | D\n")
    cat("Row 3: E full width\n")
    cat("\nFigure size:\n")
    cat(plot_width_in, " x ", round(plot_height_in, 2), " inches\n", sep = "")
    cat("\nOutput files:\n")
    print(output_status, row.names = FALSE)
    cat("============================================================\n")
    invisible(list(input_files = input_files, image_ratio = image_ratio, output_status = output_status, png = output_png, pdf = output_pdf, code_txt = output_code))
}

figure4_result <- make_figure4_composite_preserve_aspect()
