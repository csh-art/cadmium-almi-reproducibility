############################################################
## Figure 4A
## cytoHubba consensus matrix (plain, no title)
## Reproducible R code automatically exported
############################################################

make_figure4a <- function () 
{
    graphics.off()
    required_packages <- c("ggplot2", "dplyr", "tidyr")
    for (pkg in required_packages) {
        if (!requireNamespace(pkg, quietly = TRUE)) {
            install.packages(pkg, repos = "https://cloud.r-project.org", dependencies = TRUE)
        }
    }
    suppressPackageStartupMessages({
        library(ggplot2)
        library(dplyr)
        library(tidyr)
    })
    project_root <- "E:/1Data/Cd_Sarcopenia_NHANES"
    output_dir <- file.path(project_root, "new_workspace", "03_results", "M04_HubGenes_cytoHubba_GSE111016_GSE111010", "03_outputs_figures_tables", "figure")
    dir.create(output_dir, recursive = TRUE, showWarnings = FALSE)
    if (!dir.exists(output_dir)) {
        stop(paste0("\n无法创建或访问指定输出文件夹：\n", output_dir), call. = FALSE)
    }
    test_file <- file.path(output_dir, paste0(".__Figure4A_write_test_", Sys.getpid(), ".tmp"))
    write_test <- tryCatch({
        writeLines("write test", con = test_file, useBytes = TRUE)
        file.exists(test_file)
    }, error = function(e) {
        FALSE
    })
    if (file.exists(test_file)) {
        unlink(test_file)
    }
    if (!isTRUE(write_test)) {
        stop(paste0("\n指定文件夹不可写：\n", output_dir, "\n\n请检查文件夹权限。"), call. = FALSE)
    }
    presence_file_name <- "Figure_03A_cytoHubba_algorithm_presence_table.csv"
    previous_source_name <- "Figure_04A_cytoHubba_consensus_matrix_source_data.csv"
    input_candidates <- c(file.path(output_dir, presence_file_name), file.path(output_dir, previous_source_name), file.path(project_root, "99_final_confirmed", "04_figure_source_data", "Figure_03", presence_file_name), file.path(project_root, "99_final_confirmed", "04_figure_source_data", "Figure_04", presence_file_name), file.path(getwd(), presence_file_name))
    if (exists("hub_presence_df", envir = .GlobalEnv, inherits = FALSE)) {
        hub_raw <- get("hub_presence_df", envir = .GlobalEnv, inherits = FALSE)
        input_description <- "当前 R 全局环境对象：hub_presence_df"
    }
    else {
        existing_candidates <- input_candidates[file.exists(input_candidates)]
        if (length(existing_candidates) > 0) {
            input_file <- existing_candidates[1]
        }
        else {
            cat("\n首选路径中未找到输入文件，正在项目目录中自动搜索……\n")
            searched_files <- list.files(path = project_root, pattern = paste0("^", presence_file_name, "$"), recursive = TRUE, full.names = TRUE, ignore.case = TRUE)
            if (length(searched_files) == 0) {
                stop(paste0("\n没有找到 Figure 4A 输入数据。\n\n", "当前环境中不存在 hub_presence_df，项目目录中也没有找到：\n", presence_file_name, "\n\n建议将该 CSV 放入：\n", output_dir), call. = FALSE)
            }
            input_file <- searched_files[1]
        }
        hub_raw <- read.csv(input_file, stringsAsFactors = FALSE, check.names = FALSE)
        input_description <- input_file
    }
    required_columns <- c("Gene", "Degree", "MCC", "EPC")
    missing_columns <- setdiff(required_columns, colnames(hub_raw))
    if (length(missing_columns) > 0) {
        stop(paste0("\n输入数据缺少必要列：\n", paste0("  - ", missing_columns, collapse = "\n"), "\n\n当前列名：\n", paste(colnames(hub_raw), collapse = ", ")), call. = FALSE)
    }
    convert_to_binary <- function(x, column_name) {
        if (is.logical(x)) {
            result <- as.integer(x)
        }
        else {
            x_text <- toupper(trimws(as.character(x)))
            result <- rep(NA_integer_, length(x_text))
            result[x_text %in% c("1", "TRUE", "T", "YES", "Y", "SELECTED", "PRESENT", "INCLUDED")] <- 1L
            result[x_text %in% c("0", "FALSE", "F", "NO", "N", "NOT SELECTED", "ABSENT", "NOT INCLUDED")] <- 0L
            numeric_values <- suppressWarnings(as.numeric(x_text))
            numeric_positions <- is.na(result) & !is.na(numeric_values)
            result[numeric_positions] <- as.integer(numeric_values[numeric_positions] > 0)
        }
        if (any(is.na(result))) {
            invalid_values <- unique(as.character(x[is.na(result)]))
            stop(paste0("\n列 ", column_name, " 中存在无法识别的值：\n", paste0("  - ", invalid_values, collapse = "\n")), call. = FALSE)
        }
        return(result)
    }
    hub_df <- hub_raw %>% transmute(Gene = trimws(as.character(Gene)), Degree = convert_to_binary(Degree, "Degree"), MCC = convert_to_binary(MCC, "MCC"), EPC = convert_to_binary(EPC, "EPC")) %>% filter(!is.na(Gene), Gene != "") %>% group_by(Gene) %>% summarise(Degree = max(Degree, na.rm = TRUE), MCC = max(MCC, na.rm = TRUE), EPC = max(EPC, na.rm = TRUE), .groups = "drop") %>% mutate(n_algorithms = Degree + MCC + EPC) %>% filter(n_algorithms >= 1) %>% mutate(Hub_class = case_when(n_algorithms == 3 ~ 
        "Strict hub genes", n_algorithms == 2 ~ "Robust hub genes", n_algorithms == 1 ~ "Algorithm-specific genes"))
    hub_class_levels <- c("Strict hub genes", "Robust hub genes", "Algorithm-specific genes")
    hub_df <- hub_df %>% mutate(Hub_class = factor(Hub_class, levels = hub_class_levels)) %>% arrange(Hub_class, desc(Degree), desc(MCC), desc(EPC), Gene)
    number_of_genes <- nrow(hub_df)
    if (number_of_genes == 0) {
        stop("\n清洗后没有可用于绘图的基因。", call. = FALSE)
    }
    hub_df <- hub_df %>% mutate(row_number = row_number(), y_position = number_of_genes - row_number + 1, Support = paste0(n_algorithms, "/3"))
    algorithm_x <- c(Degree = 1.62, MCC = 2.43, EPC = 3.24)
    membership_long <- hub_df %>% dplyr::select(Gene, Hub_class, n_algorithms, Support, y_position, Degree, MCC, EPC) %>% pivot_longer(cols = c(Degree, MCC, EPC), names_to = "Algorithm", values_to = "Selected") %>% mutate(Algorithm = factor(Algorithm, levels = c("Degree", "MCC", "EPC")), x_position = unname(algorithm_x[as.character(Algorithm)]))
    selected_df <- membership_long %>% filter(Selected == 1)
    not_selected_df <- membership_long %>% filter(Selected == 0)
    class_blocks <- hub_df %>% group_by(Hub_class) %>% summarise(ymin = min(y_position) - 0.48, ymax = max(y_position) + 0.48, ymid = mean(y_position), .groups = "drop") %>% mutate(Class_title = case_when(as.character(Hub_class) == "Strict hub genes" ~ "Strict hub genes", as.character(Hub_class) == "Robust hub genes" ~ "Robust hub genes", TRUE ~ "Algorithm-specific"), Class_support = case_when(as.character(Hub_class) == "Strict hub genes" ~ "3/3 algorithms", as.character(Hub_class) == "Robust hub genes" ~ 
        "2/3 algorithms", TRUE ~ "1/3 algorithm"), title_y = ymid + 0.18, support_y = ymid - 0.3)
    class_sequence <- as.character(hub_df$Hub_class)
    if (number_of_genes > 1) {
        change_indices <- which(class_sequence[-number_of_genes] != class_sequence[-1])
        separator_df <- data.frame(y_position = (hub_df$y_position[change_indices] + hub_df$y_position[change_indices + 1])/2)
    }
    else {
        separator_df <- data.frame(y_position = numeric(0))
    }
    class_fill_colors <- c(`Strict hub genes` = "#E07A45", `Robust hub genes` = "#2A9D8F", `Algorithm-specific genes` = "#8996B0")
    class_border_colors <- c(`Strict hub genes` = "#A64B20", `Robust hub genes` = "#176F65", `Algorithm-specific genes` = "#56637C")
    gene_x <- 0.1
    support_x <- 4.08
    class_xmin <- 4.72
    class_xmax <- 6.12
    class_xmid <- (class_xmin + class_xmax)/2
    plot_xmin <- 0
    plot_xmax <- 6.18
    table_top_boundary <- number_of_genes + 0.5
    column_header_y <- number_of_genes + 0.92
    algorithm_line_y <- number_of_genes + 1.28
    algorithm_title_y <- number_of_genes + 1.55
    legend_y <- -0.12
    p_fig4a <- ggplot() + geom_rect(data = class_blocks, aes(ymin = ymin, ymax = ymax, fill = Hub_class), xmin = plot_xmin, xmax = class_xmin - 0.08, alpha = 0.035, color = NA) + geom_rect(data = class_blocks, aes(ymin = ymin, ymax = ymax, fill = Hub_class), xmin = class_xmin, xmax = class_xmax, alpha = 0.16, color = "white", linewidth = 0.75) + geom_segment(data = class_blocks, aes(x = class_xmin, xend = class_xmin, y = ymin + 0.03, yend = ymax - 0.03, color = Hub_class), linewidth = 3.1, lineend = "butt") + 
        geom_hline(yintercept = seq(0.5, number_of_genes + 0.5, by = 1), color = "#E4E9ED", linewidth = 0.3) + geom_hline(data = separator_df, aes(yintercept = y_position), color = "#8C979F", linewidth = 0.78) + annotate(geom = "segment", x = plot_xmin, xend = class_xmax, y = table_top_boundary, yend = table_top_boundary, color = "#4B565E", linewidth = 0.78) + annotate(geom = "segment", x = 0.91, xend = 0.91, y = 0.5, yend = number_of_genes + 1.56, color = "#D1D8DD", linewidth = 0.46) + annotate(geom = "segment", 
        x = 3.67, xend = 3.67, y = 0.5, yend = number_of_genes + 1.56, color = "#D1D8DD", linewidth = 0.46) + annotate(geom = "segment", x = 4.45, xend = 4.45, y = 0.5, yend = number_of_genes + 1.56, color = "#D1D8DD", linewidth = 0.46) + annotate(geom = "segment", x = 1.22, xend = 3.62, y = algorithm_line_y, yend = algorithm_line_y, color = "#657078", linewidth = 0.55) + geom_text(data = hub_df, aes(x = gene_x, y = y_position, label = Gene), hjust = 0, vjust = 0.5, color = "#20272D", fontface = "bold", 
        size = 3.58) + geom_point(data = not_selected_df, aes(x = x_position, y = y_position), shape = 21, size = 4.1, stroke = 0.82, fill = "white", color = "#B7C1C8") + geom_point(data = selected_df, aes(x = x_position, y = y_position, fill = Hub_class, color = Hub_class), shape = 21, size = 4.1, stroke = 0.72) + geom_text(data = hub_df, aes(x = support_x, y = y_position, label = Support), vjust = 0.5, color = "#263039", fontface = "bold", size = 3.62) + geom_text(data = class_blocks, aes(x = class_xmid, 
        y = title_y, label = Class_title, color = Hub_class), fontface = "bold", size = 3.28) + geom_text(data = class_blocks, aes(x = class_xmid, y = support_y, label = Class_support, color = Hub_class), fontface = "bold", size = 2.9) + annotate(geom = "text", x = gene_x, y = column_header_y, label = "Gene", hjust = 0, color = "#1E252B", fontface = "bold", size = 3.96) + annotate(geom = "text", x = algorithm_x["Degree"], y = column_header_y, label = "Degree", color = "#1E252B", fontface = "bold", 
        size = 3.9) + annotate(geom = "text", x = algorithm_x["MCC"], y = column_header_y, label = "MCC", color = "#1E252B", fontface = "bold", size = 3.9) + annotate(geom = "text", x = algorithm_x["EPC"], y = column_header_y, label = "EPC", color = "#1E252B", fontface = "bold", size = 3.9) + annotate(geom = "text", x = support_x, y = column_header_y, label = "Support\n(k/3)", color = "#1E252B", fontface = "bold", size = 3.48, lineheight = 0.86) + annotate(geom = "text", x = class_xmid, y = column_header_y, 
        label = "Hub class", color = "#1E252B", fontface = "bold", size = 3.9) + annotate(geom = "text", x = mean(algorithm_x), y = algorithm_title_y, label = "cytoHubba algorithm", color = "#4C565E", fontface = "bold", size = 3.58) + annotate(geom = "point", x = 1.6, y = legend_y, shape = 21, size = 3.55, stroke = 0.72, fill = class_fill_colors["Robust hub genes"], color = class_border_colors["Robust hub genes"]) + annotate(geom = "text", x = 1.78, y = legend_y, label = "Selected in top list", hjust = 0, 
        vjust = 0.5, color = "#5B656D", size = 2.96) + annotate(geom = "point", x = 3.36, y = legend_y, shape = 21, size = 3.55, stroke = 0.82, fill = "white", color = "#B7C1C8") + annotate(geom = "text", x = 3.54, y = legend_y, label = "Not selected", hjust = 0, vjust = 0.5, color = "#5B656D", size = 2.96) + scale_fill_manual(values = class_fill_colors, drop = FALSE) + scale_color_manual(values = class_border_colors, drop = FALSE) + scale_x_continuous(expand = expansion(mult = c(0, 0))) + scale_y_continuous(expand = expansion(mult = c(0, 
        0))) + coord_cartesian(xlim = c(plot_xmin, plot_xmax), ylim = c(-0.48, number_of_genes + 1.82), clip = "off") + guides(fill = "none", color = "none") + theme_void(base_size = 12, base_family = "sans") + theme(plot.margin = margin(t = 11, r = 12, b = 8, l = 12))
    print(p_fig4a)
    output_png <- file.path(output_dir, "Figure_04A_cytoHubba_consensus_matrix_plain.png")
    output_txt <- file.path(output_dir, "Figure_04A_cytoHubba_consensus_matrix_plain_code.txt")
    plot_width <- 8.9
    plot_height <- max(6.4, 0.3 * number_of_genes + 2.1)
    ggsave(filename = output_png, plot = p_fig4a, width = plot_width, height = plot_height, units = "in", dpi = 600, bg = "white", limitsize = FALSE)
    function_code <- deparse(make_figure4a, width.cutoff = 500L)
    function_code[1] <- paste0("make_figure4a <- ", function_code[1])
    complete_code <- c("############################################################", "## Figure 4A", "## cytoHubba consensus matrix (plain, no title)", "## Reproducible R code automatically exported", "############################################################", "", function_code, "", "figure4a_result <- make_figure4a()", "")
    writeLines(complete_code, con = output_txt, useBytes = TRUE)
    required_outputs <- c(output_png, output_txt)
    output_ok <- file.exists(required_outputs)
    output_sizes <- file.info(required_outputs)$size
    if (any(!output_ok) || any(is.na(output_sizes)) || any(output_sizes <= 0)) {
        failed_files <- required_outputs[!output_ok | is.na(output_sizes) | output_sizes <= 0]
        stop(paste0("\n以下文件没有成功生成：\n", paste0("  - ", failed_files, collapse = "\n")), call. = FALSE)
    }
    cat("\n", "============================================================\n", "Figure 4A 无标题版本已成功生成\n", "============================================================\n", "\n", "输入数据：\n", input_description, "\n\n", "输出文件夹：\n", normalizePath(output_dir, winslash = "/", mustWork = TRUE), "\n\n", "PNG：\n", normalizePath(output_png, winslash = "/", mustWork = TRUE), "\n\n", "完整代码 TXT：\n", normalizePath(output_txt, winslash = "/", mustWork = TRUE), 
        "\n", "============================================================\n", sep = "")
    invisible(list(plot = p_fig4a, input = input_description, output_directory = output_dir, png = output_png, code_txt = output_txt))
}

figure4a_result <- make_figure4a()
